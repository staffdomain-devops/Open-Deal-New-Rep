import re

def resolve_geo(company_props, contact_props):
  country = (company_props.get('country') or '').strip().lower()
  AU = {'australia','au'}; NZ = {'new zealand','nz'}
  US = {'united states','us','usa','united states of america'}
  UK = {'united kingdom','uk','gb','great britain'}
  if country in AU: return 'AU'
  if country in NZ: return 'NZ'
  if country in US: return 'US'
  if country in UK: return 'UK'
  phone = (contact_props.get('phone') or '').replace(' ','').replace('-','')

  if phone.startswith('+64'): return 'NZ'
  if phone.startswith('+1'): return 'US'
  if phone.startswith('+44'): return 'UK'
  return 'UNRESOLVED'
