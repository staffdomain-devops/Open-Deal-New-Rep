      Return batch.id as str.

    Function poll_batch(batch_id: str) -> object:
      while True loop:
        batch = client.messages.batches.retrieve(batch_id)
        if batch.processing_status == "ended": break
        Print status line including processing_status, request_counts.processing, request_counts.succeeded, request_counts.errored.
        time.sleep(60)
      Print: f"Batch complete: succeeded={batch.request_counts.succeeded}, errored={batch.request_counts.errored}"
      Return batch (the final MessageBatch object).

    Function process_batch_results(batch_id: str, briefs: dict) -> None:
      for item in client.messages.batches.results(batch_id):
        cid = item.custom_id
        if item.result.type == "succeeded":
          try:
            parsed = parse_output(item.result.message, cid)
            write_generated(cid, parsed, briefs[cid])
          except (MaxTokensError, OutputParseError) as exc:
            write_dlq(cid, "", "parse_output", str(exc), 0)
            print error to stderr
        else:
          write_dlq(cid, "", "batch_api_error", f"result.type={item.result.type}", 0)
          print error to stderr
