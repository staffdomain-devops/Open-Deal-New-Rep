    Soft warning functions W01–W05:

    def check_w01(generated: dict) -> tuple[bool, str]:
      combined = " ".join(generated[k]["body"] for k in EMAIL_KEYS).lower()
      for phrase in TIC_PHRASES:
        count = combined.count(phrase)
        if count > 1:
          return (True, f"W01: tic phrase '{phrase}' used {count} times across sequence")
      return (False, "")

    def check_w02(generated: dict) -> tuple[bool, str]:
      bodies = [generated[k]["body"].lower() for k in EMAIL_KEYS]
      def ngrams(text, n=4):
        words = text.split()
        return {" ".join(words[i:i+n]) for i in range(len(words) - n + 1)} if len(words) >= n else set()
      seen = {}
      for idx, body in enumerate(bodies):
        for ng in ngrams(body):
          if ng in seen:
            return (True, f"W02: 4-word phrase '{ng}' repeated across e{seen[ng]+1} and e{idx+1}")
          seen[ng] = idx
      return (False, "")

    def check_w03(generated: dict) -> tuple[bool, str]:
      if "?" not in generated["e3"]["body"]:
        return (True, "W03: e3 body missing question mark")
      return (False, "")

    def check_w04(generated: dict, brief: dict) -> tuple[bool, str]:
      for key, url_field in [("e3", "email3_url"), ("e4", "email4_url")]:
        url = brief.get(url_field, "")
        if not url: continue
        body = generated[key]["body"]
        urls = URL_REGEX.findall(body)
        if not urls: continue
        first_url_pos = body.index(urls[0])
        last_url_end = body.rindex(urls[0]) + len(urls[0])
        if first_url_pos < 20 or last_url_end > len(body) - 20:
          return (True, f"W04: {key} URL is at start or end of body (should be mid-body)")
      return (False, "")

    def check_w05(generated: dict, seen_subjects: dict, contact_id: str) -> tuple[bool, str]:
      for key in EMAIL_KEYS:
        subj = generated[key]["subject"]
        if subj in seen_subjects and seen_subjects[subj] != contact_id:
          return (True, f"W05: {key} subject '{subj}' duplicates contact {seen_subjects[subj]}")
        seen_subjects[subj] = contact_id
      return (False, "")
