    Create scripts/assemble_output.py modelled on scripts/assemble_bodies.py.

    Module docstring: "assemble_output.py — Aggregate assembled per-contact JSON files into campaign_output.json."

    Imports: json, os, sys; from utils import write_dlq.

    Constant: RUNNER_TEMP = os.environ.get("RUNNER_TEMP", ".").

    main() body steps:
    1. ids_path = os.path.join(RUNNER_TEMP, "lint_passing_ids.json").
       Try open; on FileNotFoundError print error to stderr and sys.exit(1).
       Load contact_ids list. Set total = len(contact_ids), results = [], collected_count = 0.
    2. write_dlq("batch", "", "startup", "sentinel", 0).
    3. Per-contact loop: assembled_path = os.path.join(RUNNER_TEMP, f"assembled_{cid}.json").
       Try: open + json.load; append loaded dict to results; collected_count += 1.
       Except FileNotFoundError: print SKIP {cid} to stderr; continue.
       Except Exception as exc: write_dlq(cid, "", "assemble_output", str(exc), 0);
       print ERROR to stderr; continue.
    4. out_path = os.path.join(RUNNER_TEMP, "campaign_output.json").
       Open for write; json.dump(results, f, indent=2, default=str).
    5. print(f"Output aggregation complete: {collected_count}/{total} contacts collected.")

    Guard: if __name__ == "__main__": main()
  </action>
